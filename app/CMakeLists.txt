cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(retroplc_runtime LANGUAGES C CXX)

# Suppress compiler warnings for the PLC runtime.
zephyr_compile_options(-w)

set(STRUCPP_RUNTIME_INCLUDE_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/src/strucpp-runtime")

target_sources(app PRIVATE
    src/main.cpp
    src/plc_mgmt.cpp
    src/strucpp_fault.cpp
)

target_include_directories(app PRIVATE
    "${STRUCPP_RUNTIME_INCLUDE_DIR}"
)

# The PLC project remains outside the west workspace. The IDE passes its
# generated build directory here; every C++ translation unit in that directory
# becomes part of this central Zephyr application.
set(RETROPLC_GENERATED_DIR "" CACHE PATH
    "Directory containing the C++ output generated by STruCpp")
zephyr_get(RETROPLC_GENERATED_DIR SYSBUILD LOCAL)

if(RETROPLC_GENERATED_DIR)
  if(NOT IS_ABSOLUTE "${RETROPLC_GENERATED_DIR}")
    message(FATAL_ERROR "RETROPLC_GENERATED_DIR must be an absolute path")
  endif()

  file(GLOB RETROPLC_GENERATED_SOURCES CONFIGURE_DEPENDS
      "${RETROPLC_GENERATED_DIR}/*.cpp"
      "${RETROPLC_GENERATED_DIR}/*.cc"
      "${RETROPLC_GENERATED_DIR}/*.cxx")

  file(GLOB RETROPLC_GENERATED_HEADERS CONFIGURE_DEPENDS
      "${RETROPLC_GENERATED_DIR}/*.hpp")

  if(NOT RETROPLC_GENERATED_SOURCES)
    message(FATAL_ERROR
        "No generated C++ sources found in ${RETROPLC_GENERATED_DIR}")
  endif()

  list(LENGTH RETROPLC_GENERATED_HEADERS RETROPLC_GENERATED_HEADER_COUNT)
  if(NOT RETROPLC_GENERATED_HEADER_COUNT EQUAL 1)
    message(FATAL_ERROR
        "Expected exactly one generated .hpp file in ${RETROPLC_GENERATED_DIR}")
  endif()

  list(GET RETROPLC_GENERATED_HEADERS 0 RETROPLC_GENERATED_HEADER)
  get_filename_component(RETROPLC_GENERATED_HEADER_NAME
      "${RETROPLC_GENERATED_HEADER}" NAME)

  set(RETROPLC_CONFIGURATION_CLASS "Configuration_CONFIGURATION1" CACHE STRING
      "Generated STruCpp configuration class to run")

  target_sources(app PRIVATE ${RETROPLC_GENERATED_SOURCES})
  target_include_directories(app PRIVATE "${RETROPLC_GENERATED_DIR}")
  target_compile_definitions(app PRIVATE
      RETROPLC_HAS_GENERATED_PROGRAM=1
      RETROPLC_GENERATED_HEADER=\"${RETROPLC_GENERATED_HEADER_NAME}\"
      RETROPLC_CONFIGURATION_CLASS=${RETROPLC_CONFIGURATION_CLASS}
  )
else()
  message(STATUS
      "RETROPLC_GENERATED_DIR was not supplied; building the platform shell only")
endif()
